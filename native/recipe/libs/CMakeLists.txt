cmake_minimum_required(VERSION 3.24)
project(slipstream_native_libs LANGUAGES C CXX)

include(ExternalProject)

foreach(var LAYER_PREFIX DOWNLOAD_DIR OPENSSL_TARGET SYSTEM_LIBS
		OPENSSL_URL OPENSSL_SHA256 BOOST_URL BOOST_SHA256
		LIBTORRENT_RASTERBAR_URL LIBTORRENT_RASTERBAR_SHA256)
	if(NOT DEFINED ${var})
		message(FATAL_ERROR "${var} is required; run the recipe through `release deps build`")
	endif()
endforeach()

cmake_host_system_information(RESULT jobs QUERY NUMBER_OF_LOGICAL_CORES)
set(licenses ${LAYER_PREFIX}/licenses)

ExternalProject_Add(openssl
	URL ${OPENSSL_URL}
	URL_HASH SHA256=${OPENSSL_SHA256}
	DOWNLOAD_DIR ${DOWNLOAD_DIR}
	DOWNLOAD_EXTRACT_TIMESTAMP ON
	BUILD_IN_SOURCE ON
	CONFIGURE_COMMAND ${CMAKE_COMMAND} -E env CC=${CMAKE_C_COMPILER} AR=${CMAKE_AR} RANLIB=${CMAKE_RANLIB}
		perl Configure ${OPENSSL_TARGET}
		--prefix=${LAYER_PREFIX} --libdir=lib
		no-shared no-module no-tests no-docs no-apps
		${CMAKE_C_COMPILE_OPTIONS_PIC}
	BUILD_COMMAND make -j${jobs}
	INSTALL_COMMAND make install_dev
	COMMAND ${CMAKE_COMMAND} -E copy <SOURCE_DIR>/LICENSE.txt ${licenses}/openssl.txt
)

ExternalProject_Add(boost
	URL ${BOOST_URL}
	URL_HASH SHA256=${BOOST_SHA256}
	DOWNLOAD_DIR ${DOWNLOAD_DIR}
	DOWNLOAD_EXTRACT_TIMESTAMP ON
	CONFIGURE_COMMAND ""
	BUILD_COMMAND ""
	INSTALL_COMMAND ${CMAKE_COMMAND} -E copy_directory <SOURCE_DIR>/boost ${LAYER_PREFIX}/include/boost
	COMMAND ${CMAKE_COMMAND} -E copy <SOURCE_DIR>/LICENSE_1_0.txt ${licenses}/boost.txt
)

ExternalProject_Add(libtorrent
	DEPENDS openssl boost
	URL ${LIBTORRENT_RASTERBAR_URL}
	URL_HASH SHA256=${LIBTORRENT_RASTERBAR_SHA256}
	DOWNLOAD_DIR ${DOWNLOAD_DIR}
	DOWNLOAD_EXTRACT_TIMESTAMP ON
	PATCH_COMMAND ${CMAKE_COMMAND} -DSOURCE_DIR=<SOURCE_DIR> -P ${CMAKE_CURRENT_SOURCE_DIR}/patch-libtorrent.cmake
	CMAKE_ARGS
		-DCMAKE_BUILD_TYPE=Release
		-DCMAKE_C_COMPILER=${CMAKE_C_COMPILER}
		-DCMAKE_CXX_COMPILER=${CMAKE_CXX_COMPILER}
		-DCMAKE_CXX_STANDARD=17
		-DCMAKE_POSITION_INDEPENDENT_CODE=ON
		-DCMAKE_INSTALL_PREFIX=${LAYER_PREFIX}
		-DCMAKE_INSTALL_LIBDIR=lib
		-DBUILD_SHARED_LIBS=OFF
		-Ddeprecated-functions=OFF
		-DOPENSSL_ROOT_DIR=${LAYER_PREFIX}
		-DOPENSSL_USE_STATIC_LIBS=ON
		-DBoost_INCLUDE_DIR=${LAYER_PREFIX}/include
		-DBoost_NO_BOOST_CMAKE=ON
		-DBoost_NO_SYSTEM_PATHS=ON
		-DCMAKE_POLICY_DEFAULT_CMP0167=OLD
	BUILD_COMMAND ${CMAKE_COMMAND} --build <BINARY_DIR> --parallel ${jobs}
	INSTALL_COMMAND ${CMAKE_COMMAND} --install <BINARY_DIR>
	COMMAND ${CMAKE_COMMAND} -E copy <SOURCE_DIR>/LICENSE ${licenses}/libtorrent-rasterbar.txt
)

string(REPLACE ";" "$<SEMICOLON>" system_libs "${SYSTEM_LIBS}")
add_custom_target(metadata ALL
	COMMAND ${CMAKE_COMMAND} -DLAYER_PREFIX=${LAYER_PREFIX} "-DSYSTEM_LIBS=${system_libs}"
		"-DCOMPILER=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}"
		-P ${CMAKE_CURRENT_SOURCE_DIR}/metadata.cmake
	DEPENDS libtorrent
	VERBATIM
)
